01

เส้นทางจาก Syntax สู่ API

รู้ว่าจะเรียนอะไร และทำอะไรได้เมื่อจบแต่ละช่วง

เป้าหมายของคอร์ส

Go (มักเรียกว่า Golang) เป็นภาษาที่มีชนิดข้อมูลชัดเจนและคอมไพล์เป็น executable เหมาะกับงาน backend และบริการเครือข่าย คอร์สนี้พาเขียนภาษาให้เข้าใจก่อน แล้วนำไปสร้าง Product REST API ที่เก็บข้อมูลใน MySQL จริง ใช้ standard library เป็นหลักและเพิ่ม MySQL driver เพียงตัวเดียว

ควรรู้การใช้ terminal, ไฟล์และโฟลเดอร์ และแนวคิดตัวแปร/เงื่อนไขเบื้องต้น หากยังไม่เคยเขียนโปรแกรม ให้อ่านตัวอย่างพร้อมผลลัพธ์และทำแบบฝึกหัดทีละบท ไม่จำเป็นต้องรู้ framework มาก่อน

ช่วงเนื้อหาผลงาน / เวลาฝึกโดยประมาณ
พื้นฐาน • บท 2–6ติดตั้ง, ตัวแปร, flow, collections, functionsโปรแกรมคำนวณตะกร้าสินค้า • 4–6 ชม.
ออกแบบภาษา • บท 7–12struct, interface, error, module, concurrency, testแยกโค้ดเป็น package และทดสอบได้ • 5–7 ชม.
สร้าง backend • บท 13–19HTTP, REST, MySQL, pool, CRUD, validationProduct API ครบ Create/Read/Update/Delete • 6–8 ชม.
นำไปใช้ • บท 20–24transaction, รันระบบ, ทดสอบ, debugging, capstoneตรวจความถูกต้องและต่อยอดเอง • 4–6 ชม.
เวลานี้เป็นเพียงแนวทางสำหรับผู้เรียนที่เคยเขียนโปรแกรม แนะนำแบ่งเป็น 4–6 ครั้งเรียนและเผื่อเวลาลองผิดลองถูก โค้ดบล็อกที่ระบุว่า “ส่วนหนึ่งของฟังก์ชัน” ต้องวางในฟังก์ชัน ไม่ใช่นำแต่ละบล็อกมาต่อเป็นไฟล์เดียว

ติดตั้ง Go และรันโปรแกรมแรก

เข้าใจ package main, import, func main และวงจร run → format → build

เตรียมเครื่อง

  1. ติดตั้ง Go จากเว็บไซต์ทางการ คอร์สและตัวอย่างกำหนดฐานเป็น Go 1.26+ เลือก patch ที่ยังได้รับการแก้ไขความปลอดภัยของสายรุ่นที่ใช้งาน ไม่จำเป็นต้องตรง patch กับผู้สอน
  2. ติดตั้ง editor เช่น VS Code พร้อม Go extension แล้วเปิด terminal ใหม่หลังติดตั้ง
  3. ช่วง API ใช้ MySQL 8.4 ผ่าน Docker Compose หรือ MySQL ที่ติดตั้งเอง หากใช้ Docker ทั้งชุดไม่ต้องติดตั้ง Go บนเครื่อง

  • วงเล็บปีกกาเปิดวางบรรทัดเดียวกับ func โดยปกติไม่ใส่ semicolon เอง ใช้ // สำหรับ comment หนึ่งบรรทัด และ /* ... */ สำหรับหลายบรรทัด

03

ตัวแปร ชนิดข้อมูล และ String

เลือกชนิดข้อมูลให้เหมาะ และระวังความต่างระหว่าง byte กับตัวอักษร

var, :=, const และ zero value

ส่วนหนึ่งของ func main • import fmt
var name string = "Mali"
age := 20                 // := ใช้ได้ภายในฟังก์ชันเท่านั้น
var active bool           // false
var count int             // 0
var description string    // ""
const vatRate = 0.07

age = 21                  // เปลี่ยนค่าด้วย =
fmt.Printf("%s อายุ %d active=%t\n", name, age, active)
fmt.Println(count, description, vatRate)

:= ให้ compiler อนุมานชนิดข้อมูล และต้องมีตัวแปรใหม่อย่างน้อยหนึ่งตัวใน scope เดียวกัน ห้ามประกาศซ้ำด้วย age := 21 ใน scope เดิมโดยไม่มีตัวแปรใหม่ ตัวแปร local และ import ที่ประกาศแต่ไม่ใช้ทำให้ compile ไม่ผ่าน

ชนิดใช้เมื่อข้อควรระวัง
int / int64จำนวนเต็มทั่วไป / ID และจำนวนเงินใน APIint ขนาดขึ้นกับสถาปัตยกรรม; int64 กำหนดขนาดแน่นอน
float64ค่าทศนิยมในการคำนวณทั่วไปไม่เหมาะกับการเก็บจำนวนเงินที่ต้องตรงทุกสตางค์
booltrue / falseไม่แปลงเลข 0 หรือ string เป็น bool อัตโนมัติ
stringข้อความ UTF-8 ที่แก้ byte เดิมโดยตรงไม่ได้len(s) นับ byte; range วน rune
byte / runealias ของ uint8 / int32byte ใช้ข้อมูลไบนารี; rune แทน Unicode code point

แปลงชนิดและทำงานกับข้อความ

main.go • แทนเนื้อหาไฟล์เดิมแล้ว go run .
package main

import (
    "fmt"
    "strconv"
    "strings"
    "unicode/utf8"
)

func main() {
    n, err := strconv.Atoi("42")
    if err != nil {
        fmt.Println("ไม่ใช่จำนวนเต็ม:", err)
        return
    }
    text := "Goไทย"
    fmt.Println(n + 1)                         // 43
    fmt.Println(len(text))                     // 11 bytes
    fmt.Println(utf8.RuneCountInString(text))   // 5 code points
    fmt.Println(strings.TrimSpace(" Go "))     // Go
    fmt.Println(float64(5) / 2)                // 2.5
    fmt.Println(5 / 2)                         // 2 (หารจำนวนเต็ม)
}

การ cast เป็น string(65) ได้อักขระ “A” ไม่ใช่ “65” ถ้าต้องการเลขในรูปข้อความใช้ strconv.Itoa(65) จำนวน rune ก็ยังไม่เท่ากับจำนวนตัวอักษรที่ตามองเห็นเสมอ เช่น อักษรไทยที่มีวรรณยุกต์ประกอบ

04

Operators, if, switch และ for

เขียนเงื่อนไขและทำซ้ำโดยไม่ซ่อนเส้นทางของโปรแกรม

เงื่อนไขและตัวดำเนินการ

คณิตศาสตร์ใช้ + - * / % เปรียบเทียบใช้ == != < <= > >= และตรรกะใช้ && || ! การเปรียบเทียบต้องใช้ชนิดข้อมูลที่เข้ากันได้ ไม่มีการแปลงชนิดให้อัตโนมัติแบบบางภาษา

ส่วนหนึ่งของ func main • import fmt
score := 82
if score >= 80 {
    fmt.Println("A")
} else if score >= 70 {
    fmt.Println("B")
} else {
    fmt.Println("ลองใหม่")
}

switch day := "Monday"; day {
case "Saturday", "Sunday":
    fmt.Println("วันหยุด")
default:
    fmt.Println("วันทำงาน")
}

switch ไม่ไหลต่อไป case ถัดไปโดยอัตโนมัติจึงไม่ต้องใส่ break ทุก case ตัวแปรที่ประกาศใน if หรือ switch initializer ใช้ได้เฉพาะ scope ของคำสั่งนั้น

Go มีคำสั่งวนซ้ำหลักคือ for

ส่วนหนึ่งของ func main • import fmt
for i := 0; i < 3; i++ {
    fmt.Println(i) // 0, 1, 2
}

remaining := 2
for remaining > 0 { // รูปแบบเดียวกับ while
    remaining--
}

for _, name := range []string{"Mali", "Somchai"} {
    if name == "Mali" {
        continue
    }
    fmt.Println(name) // Somchai
}

for { // วนไม่สิ้นสุดจนกว่าจะ break / return
    break
}
i++ เป็น statement ไม่ใช่ expression จึงเขียน x := i++ ไม่ได้ แบบฝึก: รับคะแนน 0–100 แล้วให้เกรด A/B/C/F และลองกรณี -1, 0, 79, 80, 100, 101
05

Array, Slice และ Map

เก็บข้อมูลหลายรายการ และเข้าใจการแชร์ backing array

Array มีขนาดตายตัว ส่วน Slice ยืดหยุ่น

ส่วนหนึ่งของ func main • import fmt
fixed := [3]int{10, 20, 30}
prices := []int{100, 200}
prices = append(prices, 300)
fmt.Println(fixed[0], len(prices), prices[1:]) // 10 3 [200 300]

original := []int{1, 2, 3}
view := original[:2]
view[0] = 99
fmt.Println(original) // [99 2 3] เพราะแชร์ backing array

cloned := make([]int, len(original))
copy(cloned, original)
cloned[0] = 0
fmt.Println(original[0]) // ยังเป็น 99

var empty []int // nil slice; len=0; append ได้
empty = append(empty, 7)
fmt.Println(empty)

len คือจำนวนสมาชิกที่เข้าถึงได้ ส่วน cap คือความจุของ backing array นับจากจุดเริ่ม slice นั้น append อาจสร้าง array ใหม่ จึงต้องรับค่าที่คืนกลับเสมอ เข้าถึง index เกินขอบเขตทำให้ panic; การ copy slice เป็น shallow copy ถ้าสมาชิกเป็น pointer ก็ยังชี้ object เดิม

Map และ comma-ok

ส่วนหนึ่งของ func main • import fmt
stock := map[string]int{"GO-1": 5}
stock["GO-2"] = 0

quantity, exists := stock["GO-2"]
fmt.Println(quantity, exists) // 0 true: มีสินค้า แต่หมด stock
quantity, exists = stock["MISSING"]
fmt.Println(quantity, exists) // 0 false: ไม่มี key

delete(stock, "GO-1")
for sku, n := range stock {
    fmt.Println(sku, n)
}

var counts map[string]int
counts = make(map[string]int) // ต้องสร้างก่อนเขียน; อ่าน nil map ได้
counts["views"]++
ลำดับการวน map ไม่แน่นอน และ map ปกติไม่ปลอดภัยสำหรับการอ่าน/เขียนพร้อมกันจากหลาย goroutine หากต้องเรียงผลลัพธ์ให้รวบรวม key และ sort หรือเรียงใน SQL ด้วย ORDER BY แบบฝึก: รวมราคาจาก slice และนับจำนวน SKU ด้วย map
06

Function, Multiple Return และ Closure

แยกงานเป็นฟังก์ชันเล็ก และคืน error อย่างชัดเจน

Parameters และผลลัพธ์หลายค่า

main.go • โปรแกรมสมบูรณ์
package main

import (
    "fmt"
    "errors"
)

func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("cannot divide by zero")
    }
    return a / b, nil
}

func sum(values ...int) int {
    total := 0
    for _, n := range values {
        total += n
    }
    return total
}

func main() {
    result, err := divide(10, 2)
    if err != nil {
        fmt.Println(err)
        return
    }
    fmt.Println(result)                 // 5
    fmt.Println(sum(1, 2, 3))           // 6
    fmt.Println(sum([]int{4, 5}...))    // 9
}

Go ส่ง argument แบบ value: ฟังก์ชันได้รับสำเนาค่า แต่ค่าของ slice/map/pointer อาจยังอ้างถึงข้อมูลภายในชุดเดียวกัน รูปแบบ (value, error) ทำให้ผู้เรียกต้องตัดสินใจเมื่อทำงานไม่สำเร็จ ไม่ควรทิ้ง error ด้วย _ โดยไม่มีเหตุผล

ฟังก์ชันเป็นค่าและสร้าง closure ได้

ประกาศนอก main • closure จดจำตัวแปร n
func counter() func() int {
    n := 0
    return func() int {
        n++
        return n
    }
}

// ภายใน main:
// next := counter()
// fmt.Println(next(), next()) // 1 2

Closure มีประโยชน์กับ callback และ middleware แต่ถ้าใช้ closure เดียวกันพร้อมกันหลาย goroutine ต้องป้องกันการเข้าถึง state เช่น n เอง แบบฝึก: เขียน totalWithDiscount(prices []int64, discount int64) (int64, error) โดยไม่ให้ส่วนลดติดลบหรือเกินยอดรวม

07

Pointer, Struct, Method และ JSON

สร้าง model ที่ใช้ต่อได้ตั้งแต่ business logic ไปจนถึง HTTP response

Pointer ชี้ไปยังค่าที่ต้องการเปลี่ยน

ประกาศ type/method นอก main
type Product struct {
    ID          int64  `json:"id"`
    Name        string `json:"name"`
    PriceSatang int64  `json:"price_satang"`
}

func (p *Product) Rename(name string) {
    p.Name = name
}

func (p Product) PriceBaht() float64 {
    return float64(p.PriceSatang) / 100
}

// ภายใน main:
// p := Product{ID: 1, Name: "Go", PriceSatang: 25000}
// ptr := &p
// ptr.Rename("Go Book")
// fmt.Println(p.Name, p.PriceBaht()) // Go Book 250
// fmt.Println((*ptr).ID)            // 1; เขียน ptr.ID ได้เช่นกัน

&x คือ address ของ x และ *ptr คือค่าที่ pointer ชี้อยู่ Receiver แบบ *Product เปลี่ยน object เดิมได้ ส่วน receiver แบบ Product ได้สำเนา struct ใช้ pointer receiver เมื่อแก้ state หรือไม่ต้องการ copy struct ใหญ่ Pointer ที่ยังไม่กำหนดค่าเป็น nil และ dereference nil จะ panic

Go ไม่มี class/inheritance แบบบางภาษา ใช้ struct, method และ composition แทน Field และชื่อฟังก์ชันที่ขึ้นต้นตัวพิมพ์ใหญ่เป็น exported; ตัวพิมพ์เล็กเข้าถึงได้ภายใน package

JSON tag เปลี่ยนชื่อ field ที่สื่อสารกับ client

ส่วนหนึ่งของฟังก์ชัน • import encoding/json และ fmt
p := Product{ID: 1, Name: "หนังสือ Go", PriceSatang: 25000}
data, err := json.Marshal(p)
if err != nil {
    return // ในโปรแกรมจริงส่งต่อหรือจัดการ error ให้เหมาะสม
}
fmt.Println(string(data))
// {"id":1,"name":"หนังสือ Go","price_satang":25000}

var decoded Product
if err := json.Unmarshal(data, &decoded); err != nil {
    return
}
encoding/json อ่าน/เขียน exported fields เท่านั้น ต้องส่ง pointer ให้ Unmarshal เพื่อแก้ค่าปลายทาง nil slice encode เป็น null; slice ว่างที่สร้างแล้ว encode เป็น [] สำหรับราคาเก็บ int64 หน่วยสตางค์: 25000 = 250.00 บาท ไม่ใช้ float64 เก็บเงินใน API นี้

Interface, Composition และ Generics

แยกสิ่งที่ต้องทำออกจากวิธีทำ เพื่อเปลี่ยนฐานข้อมูลและทดสอบได้

Interface ระบุพฤติกรรม

ประกาศนอก main • import fmt • ... ใน comment คือจุดให้จัดการ error
type Notifier interface {
    Send(message string) error
}

type ConsoleNotifier struct{}

func (ConsoleNotifier) Send(message string) error {
    fmt.Println(message)
    return nil
}

func welcome(n Notifier) error {
    return n.Send("Welcome to Go")
}

// ภายใน main: if err := welcome(ConsoleNotifier{}); err != nil { ... }

ชนิดที่มี method ครบจะ implement interface โดยอัตโนมัติ ไม่ต้องเขียน implements แนะนำเริ่มจาก interface ขนาดเล็กฝั่งผู้ใช้งาน ใน API handler ต้องการแค่พฤติกรรม List/Get/Create จึงไม่ต้องรู้วิธีเชื่อม MySQL และสามารถใช้ fake ใน test ได้

any เป็น alias ของ interface{} ใช้รับค่าได้ทุกชนิด แต่เสียการตรวจชนิดเฉพาะเจาะจง หากต้องตรวจ runtime ใช้ value, ok := x.(string) และตรวจ ok อย่าละเลย กรณี interface เก็บ typed nil pointer ตัว interface อาจไม่เท่ากับ nil

Generics ลดการเขียนฟังก์ชันซ้ำตามชนิดข้อมูล

ประกาศ generic function นอก main
func First[T any](values []T) (T, bool) {
    if len(values) == 0 {
        var zero T
        return zero, false
    }
    return values[0], true
}

// ภายใน main:
// name, ok := First([]string{"Go", "MySQL"})
// fmt.Println(name, ok) // Go true

T เป็น type parameter, any คือ constraint ที่อนุญาตทุกชนิด ถ้าต้องใช้ค่าเป็น key ของ map ต้องใช้ constraint comparable แทน ไม่จำเป็นต้องใช้ generics ทุกจุด เริ่มด้วย concrete type แล้วเพิ่มเมื่อมีความซ้ำที่ชัดเจน

ห่อ error โดยยังตรวจสาเหตุเดิมได้

ประกาศนอก main • import errors และ fmt

defer ทำงานเมื่อฟังก์ชันจบ และเรียงย้อนลำดับ

ประกาศนอก main • import fmt, os, io
func showOrder() {
    defer fmt.Println("A")
    defer fmt.Println("B")
    fmt.Println("C")
}
// showOrder() แสดง C, B, A

func readConfig(path string) ([]byte, error) {
    file, err := os.Open(path)
    if err != nil {
        return nil, fmt.Errorf("open config: %w", err)
    }
    defer file.Close() // ตัวอย่างไฟล์อ่านอย่างเดียว
    return io.ReadAll(file)
}

วาง defer หลังตรวจว่าเปิด resource สำเร็จ เช่น rows.Close, file.Close, cancel, tx.Rollback ค่าของ argument ใน defer ถูกประเมินตอนลงทะเบียน ไม่ใช่ตอนรัน และ defer ใน loop จะรอจนฟังก์ชันทั้งหมดจบ อาจถือ resource นานเกินไป

panic เหมาะกับความผิดปกติที่ไม่ควรเกิด ไม่ใช้แทน error สำหรับ JSON ผิดหรือหา row ไม่พบ recover ใช้ได้ภายใน deferred function ของ goroutine ที่ panic เท่านั้น และ os.Exit ไม่รัน defer โค้ด API จึงปิด resource ใน run() ก่อนให้ main ออกด้วย exit code

Goroutine, Channel และ Context

ทำงานพร้อมกันอย่างมีขอบเขต และหยุดงานที่ไม่ต้องการแล้ว

ส่งค่าผ่าน channel และรอให้ worker จบ

main.go • โปรแกรมสมบูรณ์
package main

import (
    "fmt"
    "sync"
)

func main() {
    results := make(chan int)
    var wg sync.WaitGroup
    for i := 1; i <= 3; i++ {
        wg.Add(1)
        go func(n int) {
            defer wg.Done()
            results <- n * n
        }(i)
    }
    go func() {
        wg.Wait()
        close(results)
    }()
    for result := range results {
        fmt.Println(result) // 1, 4, 9 แต่ลำดับไม่แน่นอน
    }
}

go f() เริ่ม goroutine; main ไม่รอ goroutine เอง Channel แบบไม่ buffer จะส่ง/รับเมื่ออีกฝั่งพร้อม การรอส่งโดยไม่มีผู้รับทำให้ค้าง ผู้ส่งหรือผู้ควบคุมผู้ส่งควรเป็นฝ่าย close channel ห้ามส่งหลัง close และห้าม close ซ้ำ การ range channel จบเมื่อ channel ถูก close และอ่านค่าที่ค้างหมดแล้ว

Context ส่งการยกเลิกและ deadline ลงไปถึง DB

ส่วนหนึ่งของฟังก์ชันที่คืน error • import context, time, fmt; parentCtx/results มาจากผู้เรียก
ctx, cancel := context.WithTimeout(parentCtx, 3*time.Second)
defer cancel()

select {
case result := <-results:
    fmt.Println(result)
case <-ctx.Done():
    return ctx.Err()
}

ใน handler เริ่มจาก r.Context() เพื่อรับการยกเลิกเมื่อ client ตัดการเชื่อมต่อ ส่ง ctx เป็น parameter แรกไปยัง QueryContext/ExecContext อย่าแทนด้วย context.Background ระหว่างทาง เพราะจะตัดสาย cancellation

HTTP server เรียก handler พร้อมกันได้อยู่แล้ว ไม่ต้องเปิด goroutine เพิ่มทุก query หากแชร์ map หรือ counter ต้องใช้ Mutex/atomic หรือออกแบบให้ worker เดียวเป็นเจ้าของ state Context ไม่ฆ่า goroutine เอง โค้ดต้องรับฟัง cancellation ด้วย ทดสอบ data race ด้วย go test -race; race detector ต้องใช้ toolchain/architecture ที่รองรับ

package main

import "testing"

func TestDivide(t *testing.T) {
    tests := []struct {
        name string
        a, b, want float64
        wantErr bool
    }{
        {"normal", 10, 2, 5, false},
        {"zero denominator", 10, 0, 0, true},
    }
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            got, err := divide(tt.a, tt.b)
            if (err != nil) != tt.wantErr {
                t.Fatalf("err=%v wantErr=%v", err, tt.wantErr)
            }
            if !tt.wantErr && got != tt.want {
                t.Errorf("got=%v want=%v", got, tt.want)
            }
        })
    }
}

เริ่มสร้าง HTTP Server

Request → Router → Handler → Response

API แรกโดยไม่ใช้ framework

main.go • โปรแกรมสมบูรณ์
package main

import (
    "encoding/json"
    "log"
    "net/http"
    "time"
)

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("GET /hello/{name}", func(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        if err := json.NewEncoder(w).Encode(map[string]string{
            "message": "Hello " + r.PathValue("name"),
        }); err != nil {
            log.Printf("encode response: %v", err)
        }
    })
    server := &http.Server{
        Addr: "127.0.0.1:8080", Handler: mux,
        ReadHeaderTimeout: 5 * time.Second,
        ReadTimeout: 5 * time.Second,
        WriteTimeout: 10 * time.Second,
        IdleTimeout: 60 * time.Second,
    }
    log.Fatal(server.ListenAndServe())
}

Routing ที่ระบุ method และ {name} ใช้ ServeMux แบบ Go 1.22 ขึ้นไป ซึ่งอยู่ในฐานเวอร์ชันของคอร์สแล้ว r.PathValue อ่าน path parameter; r.URL.Query().Get("limit") อ่าน query parameter และ r.Body อ่าน request body

ตั้ง Header ก่อน WriteHeader/Write เสมอ ถ้าเขียน body โดยไม่กำหนด status จะได้ 200 และแก้ status ภายหลังไม่ได้ GET route ของ ServeMux รองรับ HEAD ด้วย การเขียน middleware คือห่อ http.Handler แล้วเรียก next.ServeHTTP(w, r) ดูตัวอย่างจริงใน api.go

schema.sql • โค้ดที่ใช้จริง
CREATE TABLE IF NOT EXISTS products (
    id BIGINT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    sku VARCHAR(40) CHARACTER SET ascii COLLATE ascii_bin NOT NULL,
    name VARCHAR(120) NOT NULL,
    price_satang BIGINT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_products_sku UNIQUE (sku),

package main

import (
	"context"
	"database/sql"
	"fmt"
	"net"
	"os"
	"time"

	"github.com/go-sql-driver/mysql"
)

func env(key, fallback string) string {
	if value := os.Getenv(key); value != "" {
		return value
	}
	return fallback
}

func openDB(ctx context.Context) (*sql.DB, error) {
	if os.Getenv("DB_PASSWORD") == "" {
		return nil, fmt.Errorf("DB_PASSWORD is required")
	}
	cfg := mysql.NewConfig()
	cfg.User = env("DB_USER", "learner")
	cfg.Passwd = os.Getenv("DB_PASSWORD")
	cfg.Net = "tcp"
	cfg.Addr = net.JoinHostPort(env("DB_HOST", "127.0.0.1"), env("DB_PORT", "3306"))
	cfg.DBName = env("DB_NAME", "go_course")
	cfg.ParseTime = true
	cfg.Loc = time.UTC
	cfg.ClientFoundRows = true // UPDATE with unchanged values still matches one row.
	cfg.Timeout = 5 * time.Second
	cfg.ReadTimeout = 5 * time.Second
	cfg.WriteTimeout = 5 * time.Second
	cfg.Params = map[string]string{"charset": "utf8mb4", "time_zone": "'+00:00'"}
	// For a remote DB set DB_TLS=true and use a hostname matching its certificate.
	cfg.TLSConfig = os.Getenv("DB_TLS")
	db, err := sql.Open("mysql", cfg.FormatDSN())
	if err != nil {
		return nil, fmt.Errorf("open database: %w", err)
	}
	db.SetMaxOpenConns(10)
	db.SetMaxIdleConns(10)
	db.SetConnMaxLifetime(3 * time.Minute)
	db.SetConnMaxIdleTime(time.Minute)
	pingCtx, cancel := context.WithTimeout(ctx, 5*time.Second)
	defer cancel()
	if err := db.PingContext(pingCtx); err != nil {
		_ = db.Close()
		return nil, fmt.Errorf("ping database: %w", err)
	}
	return db, nil
}

	Name        string `json:"name"`
	PriceSatang int64  `json:"price_satang"`

// The HTTP layer depends on behavior, so tests can provide a fake store.
type productStore interface {
	Ping(context.Context) error
	List(context.Context, int, int) ([]Product, error)
	Get(context.Context, int64) (Product, error)
	Create(context.Context, ProductInput) (Product, error)
	Update(context.Context, int64, ProductInput) (Product, error)
	Delete(context.Context, int64) error

type mysqlStore struct{ db *sql.DB }

func (s *mysqlStore) Ping(ctx context.Context) error { return s.db.PingContext(ctx) }

func (s *mysqlStore) List(ctx context.Context, limit, offset int) ([]Product, error) {
	rows, err := s.db.QueryContext(ctx,
		`SELECT id, sku, name, price_satang, created_at FROM products ORDER BY id LIMIT ? OFFSET ?`,
		limit, offset)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	products := make([]Product, 0) // Encode an empty collection as [], not null.
	for rows.Next() {
		var p Product
		if err := rows.Scan(&p.ID, &p.SKU, &p.Name, &p.PriceSatang, &p.CreatedAt); err != nil {
			return nil, err
		}
		products = append(products, p)
	}
	return products, rows.Err()
}

func (s *mysqlStore) Get(ctx context.Context, id int64) (Product, error) {
	var p Product
	err := s.db.QueryRowContext(ctx,
		`SELECT id, sku, name, price_satang, created_at FROM products WHERE id = ?`, id).
		Scan(&p.ID, &p.SKU, &p.Name, &p.PriceSatang, &p.CreatedAt)
	return p, err
}

func (s *mysqlStore) Create(ctx context.Context, input ProductInput) (Product, error) {
	result, err := s.db.ExecContext(ctx,
		`INSERT INTO products (sku, name, price_satang) VALUES (?, ?, ?)`,
		input.SKU, input.Name, input.PriceSatang)
	if err != nil {
		return Product{}, err
	}
	id, err := result.LastInsertId()
	if err != nil {
		return Product{}, err
	}
	return s.Get(ctx, id)
}

func (s *mysqlStore) Update(ctx context.Context, id int64, input ProductInput) (Product, error) {
	result, err := s.db.ExecContext(ctx,
		`UPDATE products SET sku = ?, name = ?, price_satang = ? WHERE id = ?`,
		input.SKU, input.Name, input.PriceSatang, id)
	if err != nil {
		return Product{}, err
	}
	if err := requireRow(result); err != nil {
		return Product{}, err
	}
	return s.Get(ctx, id)
}

func (s *mysqlStore) Delete(ctx context.Context, id int64) error {
	result, err := s.db.ExecContext(ctx, `DELETE FROM products WHERE id = ?`, id)
	if err != nil {
		return err
	}
	return requireRow(result)
}

func requireRow(result sql.Result) error {
	n, err := result.RowsAffected()
	if err != nil {
		return err
	}
	if n == 0 {
		return sql.ErrNoRows
	}
	return nil
}

var _ productStore = (*mysqlStore)(nil)

func newHandler(store productStore) http.Handler {
	a := &api{store: store}
	mux := http.NewServeMux()
	mux.HandleFunc("GET /health", a.health)
	mux.HandleFunc("GET /products", a.list)
	mux.HandleFunc("GET /products/{id}", a.get)
	mux.HandleFunc("POST /products", a.create)
	mux.HandleFunc("PUT /products/{id}", a.update)
	mux.HandleFunc("DELETE /products/{id}", a.remove)
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := time.Now()
		ctx, cancel := context.WithTimeout(r.Context(), 3*time.Second)
		defer cancel()
		mux.ServeHTTP(w, r.WithContext(ctx))
		slog.Info("request", "method", r.Method, "path", r.URL.Path, "duration", time.Since(start))
	})
}

func writeJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(status)
	if body != nil {
		if err := json.NewEncoder(w).Encode(body); err != nil {
			slog.Error("encode response", "error", err)
		}
	}
}

func writeError(w http.ResponseWriter, status int, message string) {
	writeJSON(w, status, map[string]string{"error": message})
}

func storeError(w http.ResponseWriter, err error) {
	var mysqlErr *mysql.MySQLError
	switch {
	case errors.Is(err, sql.ErrNoRows):
		writeError(w, http.StatusNotFound, "product not found")
	case errors.As(err, &mysqlErr) && mysqlErr.Number == 1062:
		writeError(w, http.StatusConflict, "sku already exists")
	case errors.Is(err, context.DeadlineExceeded), errors.Is(err, context.Canceled):
		writeError(w, http.StatusGatewayTimeout, "request timed out")
	default:
		slog.Error("database operation", "error", err)
		writeError(w, http.StatusInternalServerError, "internal server error")
	}
}

func (a *api) health(w http.ResponseWriter, r *http.Request) {
	if err := a.store.Ping(r.Context()); err != nil {
		writeError(w, http.StatusServiceUnavailable, "database unavailable")
		return
	}
	writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

func queryInt(r *http.Request, key string, fallback, min, max int) (int, error) {
	values, exists := r.URL.Query()[key]
	if !exists {
		return fallback, nil
	}
	if len(values) != 1 || values[0] == "" {
		return 0, fmt.Errorf("invalid %s", key)
	}
	n, err := strconv.Atoi(values[0])
	if err != nil || n < min || n > max {
		return 0, fmt.Errorf("%s must be between %d and %d", key, min, max)
	}
	return n, nil
}

func (a *api) list(w http.ResponseWriter, r *http.Request) {

	if err != nil {
		writeError(w, 400, err.Error())
		return
	}
	offset, err := queryInt(r, "offset", 0, 0, 1_000_000)

var skuPattern = regexp.MustCompile(`^[A-Z0-9-]{1,40}$`)

func decodeInput(w http.ResponseWriter, r *http.Request) (ProductInput, bool) {
	mediaType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
	if err != nil || mediaType != "application/json" {
		writeError(w, 415, "Content-Type must be application/json")
		return ProductInput{}, false
	}
	r.Body = http.MaxBytesReader(w, r.Body, 1<<20)
	defer r.Body.Close()
	// Pointers distinguish missing/null from valid zero (a free product).
	var body struct {
		SKU         *string `json:"sku"`
		Name        *string `json:"name"`
		PriceSatang *int64  `json:"price_satang"`
	}
	decoder := json.NewDecoder(r.Body)
	decoder.DisallowUnknownFields()
	decodeErr := decoder.Decode(&body)
	if decodeErr == nil {
		var extra any
		decodeErr = decoder.Decode(&extra)
		if errors.Is(decodeErr, io.EOF) {
			decodeErr = nil
		} else if decodeErr == nil {
			decodeErr = errors.New("more than one JSON value")
		}
	}
	if decodeErr != nil {
		var tooLarge *http.MaxBytesError
		if errors.As(decodeErr, &tooLarge) {
			writeError(w, 413, "body exceeds 1 MiB")
		} else {
			writeError(w, 400, "body must contain exactly one valid product JSON object")
		}
		return ProductInput{}, false
	}
	if body.SKU == nil || body.Name == nil || body.PriceSatang == nil {
		writeError(w, 400, "sku, name and price_satang are required and cannot be null")
		return ProductInput{}, false
	}
	input := ProductInput{SKU: *body.SKU, Name: strings.TrimSpace(*body.Name), PriceSatang: *body.PriceSatang}
	if !skuPattern.MatchString(input.SKU) || input.Name == "" || utf8.RuneCountInString(input.Name) > 120 ||
		input.PriceSatang < 0 || input.PriceSatang > 1_000_000_000_000 {
		writeError(w, 400, "sku: A-Z/0-9/hyphen 1-40; name: 1-120 characters; price_satang: 0-1000000000000")
		return ProductInput{}, false
	}
	return input, true
}

func (a *api) create(w http.ResponseWriter, r *http.Request) {
	input, ok := decodeInput(w, r)
	if !ok {
		return
	}
	p, err := a.store.Create(r.Context(), input)
	if err != nil {
		storeError(w, err)
		return
	}
	w.Header().Set("Location", fmt.Sprintf("/products/%d", p.ID))
	writeJSON(w, http.StatusCreated, map[string]any{"data": p})
}

	id, ok := pathID(w, r)
	if !ok {
		return
	}
	input, ok := decodeInput(w, r)
	if !ok {
		return
	}

		storeError(w, err)
		return

package main

import (
	"context"
	"errors"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"
)

func main() {
	if err := run(); err != nil {
		slog.Error("server stopped", "error", err)
		os.Exit(1)
	}
}

func run() error {
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()
	db, err := openDB(ctx)
	if err != nil {
		return err
	}
	defer db.Close()

	srv := &http.Server{
		Addr:              env("HTTP_ADDR", "127.0.0.1:8080"),
		Handler:           newHandler(&mysqlStore{db: db}),
		ReadHeaderTimeout: 5 * time.Second,
		ReadTimeout:       5 * time.Second,
		WriteTimeout:      10 * time.Second,
		IdleTimeout:       60 * time.Second,
	}
	serverErr := make(chan error, 1)
	go func() {
		slog.Info("listening", "address", srv.Addr)
		serverErr <- srv.ListenAndServe()
	}()
	select {
	case err := <-serverErr:
		return err
	case <-ctx.Done():
		stop()
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		if err := srv.Shutdown(shutdownCtx); err != nil {
			_ = srv.Close()
			return err
		}
		if err := <-serverErr; !errors.Is(err, http.ErrServerClosed) {
			return err
		}
		return nil
	}
}

เมื่อรับ Ctrl+C หรือ SIGTERM จะหยุดรับงานใหม่และรอ request ที่ค้างภายใน 5 วินาที ถ้ายังไม่จบจะ Close server แล้วส่ง error กลับ main channel serverErr มี buffer 1 เพื่อให้ goroutine ส่งผลลัพธ์ได้แม้ main ยังอยู่ระหว่าง shutdown หลัง run จบ defer db.Close จึงปิด pool

Timeout ไม่รับประกันว่าเขียน DB ไม่สำเร็จ หาก client ไม่ได้รับ response หลัง POST ต้องตรวจผลก่อน retry การออกแบบ idempotency key เป็นวิธีต่อยอดสำหรับธุรกรรมสำคัญ

Transaction: หลายคำสั่งต้องสำเร็จด้วยกัน

BeginTx → ทำงานผ่าน tx → Commit หรือ Rollback

ตัวอย่างสร้างสินค้าสองชิ้นแบบ atomic

ฟังก์ชันเพิ่มเติม • ใช้ schema เดิม • import context, database/sql, fmt
func createPair(ctx context.Context, db *sql.DB, a, b ProductInput) error {
    tx, err := db.BeginTx(ctx, nil)
    if err != nil {
        return err
    }
    defer tx.Rollback() // หลัง Commit สำเร็จจะคืน ErrTxDone; ตั้งใจไม่ใช้ค่านี้

    for _, input := range []ProductInput{a, b} {
        _, err := tx.ExecContext(ctx,
            "INSERT INTO products (sku, name, price_satang) VALUES (?, ?, ?)",
            input.SKU, input.Name, input.PriceSatang)
        if err != nil {
            return fmt.Errorf("create pair: %w", err)
        }
    }
    return tx.Commit()
}

ถ้าชิ้นที่สอง SKU ซ้ำ การ return ทำให้ defer Rollback ยกเลิกชิ้นแรกด้วย ถ้าทุกคำสั่งผ่านแต่ Commit ล้มเหลวก็ต้องคืน error ห้ามรายงานว่าสำเร็จ ภายใน transaction ให้ใช้ tx.ExecContext/tx.QueryContext ทั้งหมด ไม่สลับไป db.ExecContext เพราะอาจใช้คนละ connection

Transaction รักษาความเป็น atomic แต่ไม่ได้แก้ทุกปัญหาการทำงานพร้อมกัน หากทำ read-modify-write ให้พิจารณา SELECT ... FOR UPDATE, atomic UPDATE หรือ version column ตามกฎของระบบ และเลือก isolation level ให้เหมาะสม

ฟังก์ชันนี้เป็นแบบฝึกเสริม ยังไม่มี endpoint ในตัวอย่าง ใช้ชื่อ SKU ใหม่สองตัวเพื่อทดลอง แล้วลองให้ตัวที่สองชน SKU เดิมและตรวจว่าตัวแรกไม่ถูกบันทึก อย่าถือ transaction ค้างระหว่างเรียก external API หรือรอ input ผู้ใช้

ฟังก์ชันเสริม • import errors
func total(prices []int64) (int64, error) {
    var sum int64
    const maxInt64 = int64(1<<63 - 1)
    for _, price := range prices {
        if price < 0 {
            return 0, errors.New("negative price")
        }
        if price > maxInt64-sum {
            return 0, errors.New("total overflow")
        }
        sum += price
    }
    return sum, nil
}

ทดสอบ [] → 0, [100,250] → 350, [-1] → error และ [MaxInt64,1] → overflow อย่าลืมว่าจำนวนเต็มมีขอบเขต แม้ใช้ int64

ทบทวนความเข้าใจ • เปิดเฉลยหลังลองตอบ

1. := ต่างจาก = อย่างไร?

:= ประกาศและกำหนดค่าภายในฟังก์ชัน โดยต้องมีตัวแปรใหม่ใน scope อย่างน้อยหนึ่งตัว; = กำหนดค่าให้ตัวแปรเดิม

2. ทำไม len("ไทย") ไม่ใช่ 3?

len(string) นับ byte ของ UTF-8 จึงได้ 9; การนับ Unicode code points ใช้ utf8.RuneCountInString และไม่เท่ากับจำนวน grapheme เสมอ

3. ทำไมแก้ slice แล้วต้นฉบับเปลี่ยน?

slice copy descriptor แต่ยังอ้าง backing array ร่วมกัน การแยกข้อมูลต้องสร้าง slice ใหม่และ copy สมาชิก

4. sql.Open สำเร็จหมายถึง DB พร้อมหรือยัง?

ยังไม่แน่นอน เพราะเป็นการเตรียม handle/pool; ต้อง PingContext เพื่อยืนยันว่าต่อได้ในขณะนั้น

5. QueryRowContext พบว่าไม่มีข้อมูลตอนไหน?

เมื่อเรียก Scan แล้วได้ sql.ErrNoRows ให้ map เป็น 404 สำหรับการอ่านสินค้าหนึ่งรายการ

6. ตรวจ SKU ก่อน INSERT เพียงอย่างเดียวพอไหม?

ไม่พอ สอง request อาจตรวจพร้อมกันแล้วต่างเห็นว่ายังไม่มี ต้องมี UNIQUE constraint และจัดการ duplicate key

7. ทำไมอัปเดตค่าเดิมอาจได้ RowsAffected=0?

MySQL ปกตินับแถวที่เปลี่ยนค่า ตัวอย่างตั้ง ClientFoundRows=true เพื่อนับแถวที่ match และไม่ส่ง 404 ผิด

8. ปิด context หมายความว่า DB ไม่ได้เขียนแน่นอนหรือไม่?

ไม่แน่นอน โดยเฉพาะเมื่อผลลัพธ์สูญหายระหว่างทาง ต้องตรวจสถานะและออกแบบ idempotency สำหรับการ retry ที่สำคัญ
