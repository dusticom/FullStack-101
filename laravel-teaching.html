Routing

กำหนด URL ว่าแต่ละ path จะไปที่ Controller ไหน

🗺️ Basic Routes (routes/web.php)

<?php
use Illuminate\Support\Facades\Route;
use App\Http\Controllers\PostController;

// ─── HTTP Verbs ───────────────────────────────────
Route::get('/hello', function() {
    return 'สวัสดี Laravel!';
});

Route::post('/submit', [FormController::class, 'store']);
Route::put('/users/{id}', [UserController::class, 'update']);
Route::delete('/users/{id}', [UserController::class, 'destroy']);

// ─── Route Parameters ─────────────────────────────
Route::get('/users/{id}', function(string $id) {
    return "User ID: {$id}";
});

// Optional parameter (?)
Route::get('/posts/{slug?}', function(string $slug = 'all') {
    return $slug;
});

// ─── Route Naming ─────────────────────────────────
Route::get('/profile', [ProfileController::class, 'show'])
    ->name('profile.show');

// ใช้ชื่อ route สร้าง URL
$url = route('profile.show');  // → /profile

// ─── Route Groups ─────────────────────────────────
Route::prefix('admin')->middleware('auth')->group(function() {
    Route::get('/dashboard', [AdminController::class, 'index']);
    Route::get('/users', [AdminController::class, 'users']);
});
// URL: /admin/dashboard, /admin/users

📦 Resource Routes (CRUD ครบ 7 routes)

/*

// ยกเว้นบาง actions

📋 Resource Controller ตัวอย่าง

<?php
namespace App\Http\Controllers;

use App\Models\Post;
use App\Http\Requests\StorePostRequest;
use Illuminate\Http\Request;
use Illuminate\Http\Response;

class PostController extends Controller
{
    // GET /posts — แสดงรายการทั้งหมด
    public function index()
    {
        $posts = Post::latest()->paginate(10);
        return view('posts.index', compact('posts'));
    }

    // GET /posts/create — แสดงฟอร์มสร้าง
    public function create()
    {
        return view('posts.create');
    }

    // POST /posts — บันทึกข้อมูลใหม่
    public function store(StorePostRequest $request)
    {
        $post = Post::create($request->validated());
        return redirect()->route('posts.show', $post)
            ->with('success', 'สร้างโพสต์เรียบร้อย!');
    }

    // GET /posts/{post} — แสดงรายการเดียว (Route Model Binding)
    public function show(Post $post)
    {
        return view('posts.show', compact('post'));
    }

    // PUT /posts/{post} — อัปเดต
    public function update(StorePostRequest $request, Post $post)
    {
        $post->update($request->validated());
        return back()->with('success', 'อัปเดตเรียบร้อย!');
    }

    // DELETE /posts/{post} — ลบ
    public function destroy(Post $post)
    {
        $post->delete();
        return redirect()->route('posts.index')
            ->with('success', 'ลบเรียบร้อย!');
    }
}
🪄
Route Model Binding — Laravel ค้นหา Model โดยอัตโนมัติ! เมื่อ type-hint Post $post ใน method Laravel จะ query Post::find($id) ให้เองโดยอัตโนมัติ ถ้าไม่พบจะ return 404

Eloquent ORM

ติดต่อฐานข้อมูลด้วย Object แทน SQL — สวยงาม ทรงพลัง

📦 สร้าง Model

# สร้าง Model
php artisan make:model Post

# สร้าง Model + Migration + Factory + Seeder + Controller
php artisan make:model Post -mfsc

# สร้าง Model + Migration + Resource Controller
php artisan make:model Post -mr

🏗️ Model ตัวอย่าง

<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\SoftDeletes;

class Post extends Model
{
    use HasFactory, SoftDeletes;

    // กำหนด fields ที่รับค่าได้ (ป้องกัน Mass Assignment)
    protected $fillable = ['title', 'body', 'user_id', 'published_at'];

    // fields ที่ซ่อนจาก JSON output
    protected $hidden = ['deleted_at'];

    // Type casting
    protected function casts(): array
    {
        return [
            'published_at' => 'datetime',
            'is_featured'  => 'boolean',
            'metadata'     => 'array',      // JSON → PHP array
        ];
    }

    // ─── Relationships ───────────────────────────────
    // Post belongsTo User
    public function user()
    {
        return $this->belongsTo(User::class);
    }

    // Post hasMany Comment
    public function comments()
    {
        return $this->hasMany(Comment::class);
    }

    // Post belongsToMany Tag (many-to-many)
    public function tags()
    {
        return $this->belongsToMany(Tag::class);
    }

    // ─── Scope — query shortcut ───────────────────────
    public function scopePublished($query)
    {
        return $query->whereNotNull('published_at')
                       ->where('published_at', '<=', now());
    }

    // Accessor — แปลงค่าตอนดึงออกมา
    public function getExcerptAttribute(): string
    {
        return str($this->body)->limit(150);
    }
}

// Mass create (ต้องมี $fillable)
$post = Post::create([
    'title' => 'Hello World',
    'body'  => 'เนื้อหา...',
    'user_id' => 1,

// สร้างแล้ว save ทีหลัง

$post->title = 'Title';
$post->save();

// ─── READ ────────────────────────────────────
// ดึงทั้งหมด

// ดึงด้วย ID
$post = Post::find(1);
$post = Post::findOrFail(1);  // → 404 ถ้าไม่เจอ

// WHERE conditions
$posts = Post::where('user_id', 1)
    ->where('published_at', '!=', null)
    ->orderBy('created_at', 'desc')
    ->limit(10)
    ->get();

// ใช้ scope
$published = Post::published()->get();

// Pagination
$posts = Post::paginate(15);
// ─── UPDATE ─────────────────────────────────
// อัปเดตผ่าน Model instance
$post->update(['title' => 'ชื่อใหม่']);

// อัปเดตตาม condition
Post::where('user_id', 1)
    ->update(['published_at' => now()]);

// ─── DELETE ─────────────────────────────────
$post->delete();                    // soft delete
$post->forceDelete();               // ลบจริง
Post::destroy([1, 2, 3]);           // ลบหลาย IDs

// ─── RELATIONSHIPS ──────────────────────────
// Eager Loading (ป้องกัน N+1 problem)

// Lazy Eager Loading

// ดึง posts พร้อม user ที่มีชื่อ 'สมชาย'
$posts = Post::whereHas('user', function($q) {
    $q->where('name', 'สมชาย');
})->get();

// Many-to-Many
$post->tags()->attach(1);
$post->tags()->detach(1);
$post->tags()->sync([1, 2, 3]);

📋 Migration File ตัวอย่าง

<?php
use Illuminate\Database\Migrations\Migration;
use Illuminate\Database\Schema\Blueprint;
use Illuminate\Support\Facades\Schema;

return new class extends Migration
{
    public function up(): void
    {
        Schema::create('posts', function (Blueprint $table) {
            $table->id();                              // bigint auto increment PK

            $table->foreignId('user_id')              // FK → users.id
                   ->constrained()
                   ->cascadeOnDelete();

            $table->string('title');                  // VARCHAR(255)
            $table->string('slug')->unique();        // unique index
            $table->text('body');                     // TEXT
            $table->longText('content')->nullable();  // LONGTEXT, null ได้
            $table->boolean('is_featured')->default(false);
            $table->integer('views')->default(0);
            $table->decimal('price', 10, 2)->nullable(); // DECIMAL(10,2)
            $table->json('metadata')->nullable();    // JSON column
            $table->enum('status', ['draft','published','archived'])
                   ->default('draft');
            $table->timestamp('published_at')->nullable();

            $table->softDeletes();                    // deleted_at column
            $table->timestamps();                     // created_at, updated_at

            // Indexes
            $table->index(['user_id', 'status']);
            $table->fullText(['title', 'body']);      // Full-text search
        });
    }

    public function down(): void
    {
        Schema::dropIfExists('posts');
    }
};

Middleware

🔧 สร้างและใช้ Middleware

# สร้าง Middleware
php artisan make:middleware EnsureUserIsAdmin
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureUserIsAdmin
{
    public function handle(Request $request, Closure $next): Response
    {
        // ตรวจสอบก่อนถึง Controller
        if (! $request->user() || $request->user()->role !== 'admin') {
            return redirect('/home')->with('error', 'ไม่มีสิทธิ์');
        }

        $response = $next($request);  // ส่งต่อไป Controller

        // ทำอะไรบางอย่างหลัง Response ก็ได้
        $response->headers->set('X-Powered-By', 'Laravel');

        return $response;
    }
}
// ใน routes/web.php
Route::get('/admin', [AdminController::class, 'index'])
    ->middleware('auth')
    ->middleware('admin');

// หรือจัดกลุ่ม
Route::middleware(['auth', 'admin'])
    ->prefix('admin')
    ->group(function () {
        Route::get('/dashboard', ...);
        Route::resource('users', ...);
    });
// Middleware ในตัว Laravel
'auth'          // ต้อง login
'auth:sanctum'  // ต้มี API token
'guest'         // ต้องยัง login
'verified'      // ต้อง verify email
'throttle:60,1' // rate limit 60/นาที
'signed'        // ต้องมี signed URL

🛡️ Form Request Validation

# สร้าง Form Request
php artisan make:request StorePostRequest
<?php
namespace App\Http\Requests;

class StorePostRequest extends FormRequest
{
    // กำหนดว่าใครมีสิทธิ์ submit
    public function authorize(): bool
    {
        return $this->user()->can('create', Post::class);
    }

    public function rules(): array
    {
        return [
            'title'    => ['required', 'string', 'max:255'],
            'slug'     => ['required', 'unique:posts,slug'],
            'body'     => ['required', 'string', 'min:10'],
            'status'   => ['required', 'in:draft,published'],
            'category_id' => ['required', 'exists:categories,id'],
            'tags'     => ['nullable', 'array'],
            'tags.*'   => ['exists:tags,id'],
            'image'    => ['nullable', 'image', 'max:2048'],
            'price'    => ['nullable', 'numeric', 'min:0'],
        ];
    }

    // Custom error messages
    public function messages(): array
    {
        return [
            'title.required' => 'กรุณาใส่ชื่อโพสต์',
            'body.min'       => 'เนื้อหาต้องมีอย่างน้อย 10 ตัวอักษร',
        ];
    }
}

🔒 Sanctum API Token

composer require laravel/sanctum

php artisan migrate

# เพิ่ม HasApiTokens ใน User Model
use Laravel\Sanctum\HasApiTokens;

class User extends Authenticatable
{
    use HasApiTokens, HasFactory, Notifiable;
}
// API Login — สร้าง Token
public function login(Request $request)
{
    if (! auth()->attempt($request->only('email', 'password'))) {
        return response()->json([
            'message' => 'Unauthorized'
        ], 401);
    }

    $user  = auth()->user();
    $token = $user->createToken('auth-token')->plainTextToken;

    return response()->json([
        'user'  => $user,
        'token' => $token,
    ]);
}

// API Logout — เพิกถอน Token
public function logout(Request $request)
{
    $request->user()->currentAccessToken()->delete();
    return response()->json(['message' => 'Logged out']);
}

<?php
// app/Http/Resources/PostResource.php
class PostResource extends JsonResource
{
    public function toArray(Request $request): array
    {
        return [
            'id'         => $this->id,
            'title'      => $this->title,
            'excerpt'    => str($this->body)->limit(100),
            'status'     => $this->status,
            'created_at' => $this->created_at->toIso8601String(),
            'author'     => new UserResource($this->whenLoaded('user')),
            'comments_count' => $this->whenCounted('comments'),
            'url'        => route('posts.show', $this->id),
        ];
    }
}

// ใช้ใน Controller
public function show(Post $post)
{
    return new PostResource($post->load('user'));
}

public function index()
{
    $posts = Post::with('user')
        ->withCount('comments')
        ->latest()
        ->paginate(15);

    return PostResource::collection($posts);
}

📦 Job Class

<?php
namespace App\Jobs;

use App\Models\User;
use App\Mail\WelcomeMail;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Support\Facades\Mail;

class SendWelcomeEmail implements ShouldQueue
{
    use Dispatchable, Queueable;

    public int $tries = 3;        // ลองใหม่ 3 ครั้ง
    public int $timeout = 60;    // timeout 60 วินาที

    public function __construct(
        private readonly User $user
    ) {}

    public function handle(): void
    {
        Mail::to($this->user)
            ->send(new WelcomeMail($this->user));
    }

    // เรียกเมื่อ job ล้มเหลวทุก retry
    public function failed(\Throwable $e): void
    {
        // Log, แจ้งเตือน admin ฯลฯ
    }
}

🚀 Dispatch Job

// ส่งไปทำงานใน Queue
SendWelcomeEmail::dispatch($user);

// ส่งหน่วงเวลา 5 นาที
SendWelcomeEmail::dispatch($user)
    ->delay(now()->addMinutes(5));

// ส่งไปที่ queue เฉพาะ
SendWelcomeEmail::dispatch($user)
    ->onQueue('emails');

// ทำทันที (ไม่ผ่าน Queue)
SendWelcomeEmail::dispatchSync($user);

// รัน Worker ประมวลผล Queue
php artisan queue:work

# ระบุ connection และ queue
php artisan queue:work redis --queue=emails,default

# รัน job เดียวแล้วหยุด
php artisan queue:work --once

# ดู status
php artisan queue:monitor
php artisan queue:failed
