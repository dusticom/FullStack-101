}

// src/store/features/counterSlice.ts
import { createSlice, PayloadAction } from '@reduxjs/toolkit'

interface CounterState {
  value: number
  status: 'idle' | 'loading' | 'failed'
}

const initialState: CounterState = {
  value: 0,
  status: 'idle',
}

export const counterSlice = createSlice({
  name: 'counter',
  initialState,
  reducers: {
    // Immer ทำให้เขียน "mutate" ได้ แต่จริงๆ immutable
    increment: (state) => {
      state.value += 1
    },
    decrement: (state) => {
      state.value -= 1
    },
    incrementByAmount: (state, action: PayloadAction<number>) => {
      state.value += action.payload
    },
    reset: (state) => {
      state.value = 0
    },
  },
})

// Export actions
export const { increment, decrement, incrementByAmount, reset } =
  counterSlice.actions

// Export reducer
export default counterSlice.reducer

// Selectors
export const selectCount = (state: RootState) => state.counter.value

⚡ createAsyncThunk — Async Actions

// src/store/features/userSlice.ts
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit'

interface User { id: number; name: string; email: string }
interface UserState {
  users: User[]
  loading: boolean
  error: string | null
}

// Async thunk — fetch users จาก API
export const fetchUsers = createAsyncThunk(
  'users/fetchAll',
  async (_, { rejectWithValue }) => {
    try {
      const res = await fetch('/api/users')
      if (!res.ok) throw new Error('Network error')
      return (await res.json()) as User[]
    } catch (err) {
      return rejectWithValue((err as Error).message)
    }
  }
)

const userSlice = createSlice({
  name: 'users',
  initialState: { users: [], loading: false, error: null } as UserState,
  reducers: {},
  extraReducers: (builder) => {
    builder
      .addCase(fetchUsers.pending, (state) => {
        state.loading = true; state.error = null
      })
      .addCase(fetchUsers.fulfilled, (state, { payload }) => {
        state.loading = false; state.users = payload
      })
      .addCase(fetchUsers.rejected, (state, { payload }) => {
        state.loading = false
        state.error = payload as string
      })
  },
})

export default userSlice.reducer

🛒 ตัวอย่างจริง: Cart Slice

// src/store/features/cartSlice.ts
import { createSlice, PayloadAction } from '@reduxjs/toolkit'

interface CartItem {
  id: number; name: string; price: number; qty: number
}
interface CartState { items: CartItem[] }

const cartSlice = createSlice({
  name: 'cart',
  initialState: { items: [] } as CartState,
  reducers: {
    addItem: (state, { payload }: PayloadAction<CartItem>) => {
      const existing = state.items.find((i) => i.id === payload.id)
      if (existing) existing.qty += 1
      else state.items.push({ ...payload, qty: 1 })
    },
    removeItem: (state, { payload }: PayloadAction<number>) => {
      state.items = state.items.filter((i) => i.id !== payload)
    },
    updateQty: (state, { payload }: PayloadAction<{ id: number; qty: number }>) => {
      const item = state.items.find((i) => i.id === payload.id)
      if (item) item.qty = payload.qty
    },
    clearCart: (state) => { state.items = [] },
  },
})

export const { addItem, removeItem, updateQty, clearCart } = cartSlice.actions
export const selectCartTotal = (state: RootState) =>
  state.cart.items.reduce((sum, i) => sum + i.price * i.qty, 0)
export default cartSlice.reducer

🚀 RTK Query — Data Fetching + Caching

// src/store/services/apiSlice.ts
import { createApi, fetchBaseQuery } from '@reduxjs/toolkit/query/react'

interface Post { id: number; title: string; body: string }

export const apiSlice = createApi({
  reducerPath: 'api',
  baseQuery: fetchBaseQuery({ baseUrl: '/api' }),
  tagTypes: ['Post', 'User'],
  endpoints: (builder) => ({
    // GET /api/posts
    getPosts: builder.query<Post[], void>({
      query: () => '/posts',
      providesTags: ['Post'],
    }),
    // GET /api/posts/:id
    getPost: builder.query<Post, number>({
      query: (id) => `/posts/${id}`,
    }),
    // POST /api/posts
    createPost: builder.mutation<Post, Omit<Post, 'id'>>({
      query: (body) => ({ url: '/posts', method: 'POST', body }),
      invalidatesTags: ['Post'],  // auto refetch
    }),
    // DELETE /api/posts/:id
    deletePost: builder.mutation<void, number>({
      query: (id) => ({ url: `/posts/${id}`, method: 'DELETE' }),
      invalidatesTags: ['Post'],
    }),
  }),
})

export const {
  useGetPostsQuery,
  useGetPostQuery,
  useCreatePostMutation,
  useDeletePostMutation,
} = apiSlice

// เพิ่ม api reducer ใน store
// reducer: { ..., [apiSlice.reducerPath]: apiSlice.reducer }
// middleware: (getDefaultMiddleware) => getDefaultMiddleware().concat(apiSlice.middleware)

💡 ใช้ RTK Query ใน Component

"use client"
import { useGetPostsQuery, useDeletePostMutation } from '@/store/services/apiSlice'

export default function PostList() {
  // Auto fetch, cache, loading/error states
  const { data: posts, isLoading, isError, refetch } = useGetPostsQuery()
  const [deletePost, { isLoading: isDeleting }] = useDeletePostMutation()

  if (isLoading) return <p>กำลังโหลด...</p>
  if (isError)   return <p>เกิดข้อผิดพลาด <button onClick={refetch}>ลองใหม่</button></p>

  return (
    <ul>
      {posts?.map((post) => (
        <li key={post.id}>
          {post.title}
          <button
            onClick={() => deletePost(post.id)}
            disabled={isDeleting}
          >
            ลบ
          </button>
        </li>
      ))}
    </ul>
  )
}

✅ ทำ

  • ใช้ useAppDispatch / useAppSelector แทน hooks ทั่วไป
  • แยก selector ออกมาเป็น function ต่างหาก
  • ใช้ RTK Query แทนการ fetch เองเสมอเมื่อทำได้
  • กำหนด TypeScript types ทุกที่
  • ใช้ Server Components สำหรับ static data
  • เก็บ RootState type ไว้ที่เดียว
  • ใช้ createEntityAdapter สำหรับ list data

❌ หลีกเลี่ยง

  • ไม่เก็บ derived state ที่คำนวณได้จาก state อื่น
  • ไม่ dispatch action ใน reducer
  • ไม่ทำ async ใน reducer (ใช้ thunk แทน)
  • ไม่ mutate state โดยตรง (นอก createSlice)
  • ไม่ใส่ non-serializable values ใน store
  • ไม่ fetch ข้อมูลใน Server Component แล้วส่งผ่าน Redux
  • ไม่ใช้ any type ใน TypeScript

🔐 Custom Hook Pattern

// hooks/useCart.ts — ห่อ logic ไว้ที่เดียว
"use client"
import { useAppDispatch, useAppSelector } from '@/store'
import { addItem, removeItem, clearCart, selectCartTotal } from '@/store/features/cartSlice'
import type { CartItem } from '@/types'

export function useCart() {
  const dispatch = useAppDispatch()
  const items    = useAppSelector(s => s.cart.items)
  const total    = useAppSelector(selectCartTotal)

  return {
    items,
    total,
    itemCount: items.length,
    addItem:   (item: CartItem) => dispatch(addItem(item)),
    removeItem:(id: number)    => dispatch(removeItem(id)),
    clearCart: ()               => dispatch(clearCart()),
  }
}

// ใช้งาน — สะอาดกว่ามาก
const { items, total, addItem } = useCart()

การจัด Style ใน Next.js

ตั้งแต่ CSS พื้นฐานจนถึง Tailwind, Sass และ CSS-in-JS

📌
ตัวอย่างใช้ Next.js 16, App Router และ TypeScript โดยมี src/app/ ตามโครงสร้างในบทที่ 3 แต่ละตัวอย่างเป็นทางเลือกแยกกัน สามารถผสมวิธีที่เหมาะกับงานได้

🧭 เลือกวิธีไหนดี?

วิธีเหมาะกับขอบเขต / สิ่งที่ต้องเตรียม
Global CSS / CSS ภายนอกพื้นฐานทั้งเว็บ, reset, design tokens หรือ CSS จากแพ็กเกจกระทบทั้งเว็บ ต้องระวังชื่อชนกัน
CSS Modulesสไตล์เฉพาะ componentชื่อ class ถูกจำกัดขอบเขตอัตโนมัติ ไม่ต้องติดตั้งเพิ่ม
Inline styleค่าที่เปลี่ยนตาม props หรือข้อมูลอยู่บน element; ไม่รองรับ pseudo-class หรือ media query โดยตรง
Tailwind CSSประกอบหน้าด้วย utility classesต้องตั้งค่า Tailwind; template ที่เลือก Tailwind มีให้แล้ว
Sass / SCSSต้องการตัวแปร, nesting และ mixin ตอนคอมไพล์ติดตั้ง sass; ใช้ร่วมกับ CSS Modules ได้
CSS-in-JSสไตล์ที่ผูกกับ component และ propsตรวจสอบการรองรับ App Router และการตั้งค่า SSR ของแต่ละไลบรารี

1. Global CSS — ใช้ร่วมกันทั้งแอป

ใส่กฎพื้นฐานและ CSS variables ไว้ใน globals.css แล้ว import ใน root layout ซึ่งครอบทุกหน้า กฎเหล่านี้เป็น global จึงควรตั้งชื่อให้ชัดเจน

/* src/app/globals.css */
:root { --brand: #2563eb; --surface: #f3f4f6; }
* { box-sizing: border-box; }
body { margin: 0; font-family: sans-serif; background: var(--surface); }
a { color: var(--brand); }
.site-title { color: var(--brand); }
// src/app/layout.tsx
import './globals.css'
import type { ReactNode } from 'react'

export default function RootLayout({ children }: { children: ReactNode }) {
  return <html lang="th"><body>{children}</body></html>
}
// src/app/page.tsx
export default function Home() {
  return <h1 className="site-title">สวัสดี Next.js</h1>
}

สำหรับ Pages Router ให้นำเข้า global CSS ใน pages/_app.tsx แทน หากเปิด Tailwind อยู่ ให้คง @import "tailwindcss"; ในไฟล์ global ด้วย

2. CSS Modules — แยก style ราย component

ตั้งชื่อไฟล์ลงท้าย .module.css แล้ว import เป็น object Next.js จะสร้างชื่อ class เฉพาะ ช่วยลดปัญหาชื่อชนกัน ใช้ได้ทั้ง Server และ Client Components

/* src/components/Card.module.css */
.card { padding: 24px; border-radius: 12px; background: white; }
.title { color: #1d4ed8; }
.card:hover { box-shadow: 0 4px 16px #0002; }
@media (max-width: 600px) { .card { padding: 16px; } }
// src/components/Card.tsx
import styles from './Card.module.css'

export default function Card({ title }: { title: string }) {
  return <article className={styles.card}>
    <h2 className={styles.title}>{title}</h2>
  </article>
}
// src/app/page.tsx
import Card from '@/components/Card'
export default function Home() {
  return <Card title="เรียนรู้ CSS Modules" />
}

CSS Modules จำกัดชื่อ class ไม่ได้แยก DOM หรือป้องกันการสืบทอด CSS ทั้งหมด

3. Inline style และ class แบบมีเงื่อนไข

ใช้ object กับ style: ชื่อ property เป็น camelCase เช่น backgroundColor และตัวเลขส่วนใหญ่มีหน่วย px โดยอัตโนมัติ เหมาะกับค่าที่ได้จากข้อมูล

// src/components/Progress.tsx
export default function Progress({ value }: { value: number }) {
  const percent = Math.max(0, Math.min(100, value))
  return <div role="progressbar" aria-label="ความคืบหน้า"
    aria-valuenow={percent} aria-valuemin={0} aria-valuemax={100}>
    <div style={{ width: `${percent}%`, backgroundColor: '#2563eb', height: 12 }} />
  </div>
}
// เลือก class ตาม props (ใช้ร่วมกับ CSS Modules ได้)
import styles from './Button.module.css'
export function Button({ primary }: { primary: boolean }) {
  const className = `${styles.button} ${primary ? styles.primary : ''}`
  return <button className={className}>บันทึก</button>
}
/* src/components/Button.module.css */
.button { padding: 8px 16px; border: 1px solid #555; }
.primary { background: #2563eb; color: white; }

สำหรับ hover, focus และ responsive ให้ใช้ CSS หรือ utility classes ควบคู่กัน; ถ้าต้องตอบสนองต่อ state หรือ event ให้ประกาศ 'use client' ในไฟล์ component นั้น

4. Tailwind CSS — utility classes

หากสร้างโปรเจกต์ด้วย --tailwind แล้วข้ามขั้นติดตั้งได้ ตัวอย่างนี้ใช้ Tailwind CSS v4; ในโปรเจกต์ที่ยังไม่มี ให้ตั้งค่าดังนี้

npm install -D tailwindcss @tailwindcss/postcss postcss
// postcss.config.mjs
export default { plugins: { '@tailwindcss/postcss': {} } }
/* src/app/globals.css */
@import "tailwindcss";

ตรวจสอบว่า root layout import ./globals.css แล้ว จากนั้นใช้ utility classes ใน JSX ได้ทั้ง Server และ Client Components

// src/app/page.tsx
export default function Home() {
  return <main className="mx-auto max-w-xl p-6">
    <h1 className="text-3xl font-bold text-blue-700">สวัสดี Tailwind</h1>
    <p className="mt-2 text-gray-600">ปรับหน้าจอและสถานะด้วย utility</p>
    <button className="mt-4 rounded bg-blue-600 px-4 py-2 text-white hover:bg-blue-700 focus-visible:outline-2 focus-visible:outline-offset-2 sm:px-6">
      เริ่มเรียน
    </button>
  </main>
}

hover: ใช้กับสถานะเมาส์ และ sm: เริ่มทำงานที่ breakpoint ขนาดเล็กขึ้นไป เขียนชื่อ class แบบเต็ม เช่น active ? 'bg-blue-600' : 'bg-gray-600' เพื่อให้ Tailwind ตรวจพบ อย่าสร้างชื่อแบบ `bg-${color}-600`

7. CSS-in-JS — styled-components

ตัวอย่างใช้ styled-components v6 ใน Client Component พร้อม registry สำหรับ SSR และ streaming โดยให้รวมการตั้งค่านี้กับไฟล์เดิมของโปรเจกต์

npm install styled-components
// next.config.ts
import type { NextConfig } from 'next'
const nextConfig: NextConfig = { compiler: { styledComponents: true } }
export default nextConfig
// src/lib/StyledComponentsRegistry.tsx
'use client'
import { useState, type ReactNode } from 'react'
import { useServerInsertedHTML } from 'next/navigation'
import { ServerStyleSheet, StyleSheetManager } from 'styled-components'

export default function StyledComponentsRegistry({ children }: { children: ReactNode }) {
  const [sheet] = useState(() => new ServerStyleSheet())
  useServerInsertedHTML(() => {
    const styles = sheet.getStyleElement()
    sheet.instance.clearTag()
    return <>{styles}</>
  })
  if (typeof window !== 'undefined') return <>{children}</>
  return <StyleSheetManager sheet={sheet.instance}>{children}</StyleSheetManager>
}
// src/app/layout.tsx
import './globals.css'
import type { ReactNode } from 'react'
import StyledComponentsRegistry from '@/lib/StyledComponentsRegistry'
export default function RootLayout({ children }: { children: ReactNode }) {
  return <html lang="th"><body>
    <StyledComponentsRegistry>{children}</StyledComponentsRegistry>
  </body></html>
}
// src/components/SaveButton.tsx
'use client'
import styled from 'styled-components'
const Button = styled.button`
  padding: 0.5rem 1rem;
  color: white;
  background: #2563eb;
  &:hover { background: #1d4ed8; }
`
export default function SaveButton() { return <Button>บันทึก</Button> }

8. ฟอนต์, responsive และธีม

next/font ช่วยจัดการฟอนต์และใช้ร่วมกับวิธีจัด style ข้างต้นได้ ตัวอย่างใช้ฟอนต์ที่รองรับภาษาไทย

ตัวอย่างธีมตามการตั้งค่าระบบและ responsive CSS สามารถอยู่ใน global CSS หรือ CSS Module ได้

  :root { --bg: #111827; --text: #f9fafb; }

การเปลี่ยนธีมด้วยปุ่มต้องจัดการ state เพิ่มเติม; ตรวจสีและ focus ให้มองเห็นได้ทั้งสองธีม และรัน npm run build เพื่อตรวจลำดับ CSS ใน production

📝 แบบฝึกหัด

  1. สร้างหน้าโปรไฟล์ที่ใช้ global CSS กำหนดสีพื้นฐานและ CSS Module สำหรับการ์ด
  2. ทำปุ่มสองสถานะด้วย conditional class แล้วเพิ่ม hover และ focus-visible
  3. สร้างการ์ดเดียวกันด้วย Tailwind จากนั้นปรับให้แสดงผลได้ทั้งมือถือและเดสก์ท็อป
  4. เพิ่ม SCSS variable หรือธีมตามระบบ แล้วตรวจหน้าด้วย npm run build
