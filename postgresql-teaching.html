บทที่ 1 — รู้จัก PostgreSQL

ฐานข้อมูลเชิงสัมพันธ์และแนวคิดที่ควรรู้

PostgreSQL คืออะไร

PostgreSQL เป็นระบบจัดการฐานข้อมูลเชิงสัมพันธ์แบบโอเพนซอร์ส รองรับ SQL, transaction, constraints, JSONB และ index หลายชนิด เหมาะกับการเก็บข้อมูลแอปพลิเคชันที่ต้องการความถูกต้องและ query ที่ซับซ้อน

ผู้เริ่มต้นควรรู้จักตาราง แถว คีย์ และ SQL พื้นฐานก่อน หากยังไม่คุ้นเคย เริ่มได้ที่ SQL Fundamentals บทนี้ใช้ PostgreSQL 16 ขึ้นไปและตัวอย่างร้านค้าเดียวกันตลอดบทเรียน

โครงสร้างที่พบ

Server / cluster → database → schema → table โดย schema เป็น namespace ภายใน database; โดยทั่วไปตารางเริ่มต้นอยู่ใน schema public ส่วน role คือบัญชีหรือกลุ่มสิทธิ์ที่ใช้เชื่อมต่อและจัดการข้อมูล

บทที่ 2 — ติดตั้งและเชื่อมต่อ

เริ่มฝึกด้วย Docker หรือ PostgreSQL ที่ติดตั้งเอง

เริ่มด้วย Docker

ติดตั้ง Docker และเปิด Docker Engine ก่อน คำสั่งนี้ใช้รหัสผ่านสำหรับเครื่องฝึกเท่านั้น และเก็บข้อมูลไว้ใน named volume:

Terminal • macOS / Linux / Bash
docker run --name pg-learn -e POSTGRES_PASSWORD=learn_only \
  -e POSTGRES_DB=shop -p 127.0.0.1:5432:5432 \
  -v pg-learn-data:/var/lib/postgresql/data -d postgres:16

docker exec -it pg-learn psql -U postgres -d shop

รอให้ server เริ่มทำงานก่อนเชื่อมต่อ หาก port 5432 ถูกใช้ ให้เปลี่ยนด้านซ้ายของ mapping เช่น 127.0.0.1:5433:5432 หรือใช้ PostgreSQL ที่ติดตั้งเอง แล้วเชื่อมต่อด้วย psql -h localhost -U postgres -d shop

คำสั่ง psql

คำสั่งขึ้นต้นด้วย backslash เป็นคำสั่งของ psql ไม่ใช่ SQL และไม่ต้องใส่ semicolon:

psql
\conninfo       -- ดู connection ปัจจุบัน
\l              -- รายชื่อ database
\dt             -- ตารางใน schema ที่มองเห็น
\d products     -- โครงสร้างตาราง (หลังสร้างแล้ว)
\q              -- ออกจาก psql

คำสั่ง SQL ต้องจบด้วย ; ส่วนผลที่ขึ้นกับข้อมูลจริงอาจต่างจากตัวอย่าง

บทที่ 3 — สร้าง schema ร้านค้า

ชนิดข้อมูลและข้อกำหนดความถูกต้อง

ตารางสินค้าและคำสั่งซื้อ

รัน SQL ต่อไปนี้ครั้งเดียวใน database shop โดยเริ่มจากตารางที่ถูกอ้างอิงก่อน:

SQL
CREATE TABLE products (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  sku text NOT NULL UNIQUE,
  name text NOT NULL,
  price numeric(10, 2) NOT NULL CHECK (price >= 0),
  stock integer NOT NULL DEFAULT 0 CHECK (stock >= 0),
  metadata jsonb NOT NULL DEFAULT '{}'::jsonb,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE orders (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  product_id bigint NOT NULL REFERENCES products(id),
  quantity integer NOT NULL CHECK (quantity > 0),
  created_at timestamptz NOT NULL DEFAULT now()
);

identity สร้างเลข ID, numeric เหมาะกับเงินที่ต้องการความแม่นยำทศนิยม, timestamptz เก็บเวลาที่มีความหมายเป็นจุดเวลา และ foreign key ป้องกัน order ที่อ้างสินค้าที่ไม่มีอยู่

บทที่ 4 — เพิ่ม อ่าน แก้ไข ลบ

ทำ CRUD พร้อม RETURNING และ ON CONFLICT

เพิ่มข้อมูลและอ่านกลับ

SQL
INSERT INTO products (sku, name, price, stock, metadata)
VALUES
  ('BK-01', 'PostgreSQL Basics', 450.00, 10, '{"category":"book"}'),
  ('PN-01', 'Pen', 25.00, 30, '{"category":"stationery"}')
RETURNING id, sku, name;

SELECT id, sku, name, price, stock
FROM products
WHERE price >= 25
ORDER BY price DESC, id
LIMIT 10;

RETURNING ส่งค่าจากแถวที่แก้ไขกลับมาได้ทันที และ ORDER BY ทำให้ลำดับผลลัพธ์ชัดเจน

แก้ไข ลบ และ upsert

SQL
UPDATE products
SET price = 475.00
WHERE sku = 'BK-01'
RETURNING id, price;

INSERT INTO products (sku, name, price, stock)
VALUES ('PN-01', 'Blue Pen', 30.00, 20)
ON CONFLICT (sku) DO UPDATE
SET name = EXCLUDED.name, price = EXCLUDED.price
RETURNING id, sku, name, price;

-- ตัวอย่างลบสินค้าที่ไม่มีอยู่ จึงไม่กระทบข้อมูลฝึก
DELETE FROM products WHERE sku = 'NOT-FOUND' RETURNING id;

ตรวจ WHERE ด้วย SELECT ก่อน UPDATE หรือ DELETE; การลบสินค้าที่มี order อ้างอยู่จะถูก foreign key ปฏิเสธตามค่าเริ่มต้น

บทที่ 5 — JOIN และ Aggregate

เชื่อมข้อมูลและสรุปผล

ตัวอย่าง JOIN

SQL
INSERT INTO orders (product_id, quantity)
SELECT id, 2 FROM products WHERE sku = 'BK-01';

SELECT o.id AS order_id, p.name, o.quantity,
       p.price * o.quantity AS current_price_total
FROM orders AS o
JOIN products AS p ON p.id = o.product_id
ORDER BY o.id;

ตัวอย่างคำนวณจากราคาปัจจุบัน ในระบบจริงควรเก็บราคาขณะซื้อใน order item เพื่อให้ประวัติไม่เปลี่ยนเมื่อแก้ราคาสินค้า

รวมจำนวนตามสินค้า

SQL
SELECT p.sku, p.name, COALESCE(SUM(o.quantity), 0) AS units
FROM products AS p
LEFT JOIN orders AS o ON o.product_id = p.id
GROUP BY p.id, p.sku, p.name
ORDER BY units DESC, p.id;

LEFT JOIN เก็บสินค้าที่ยังไม่มี order และ COALESCE แทนค่า NULL ด้วย 0

บทที่ 6 — ความสามารถเฉพาะ PostgreSQL

JSONB, ILIKE และ CTE

ค้นหา JSONB

SQL
SELECT sku, metadata ->> 'category' AS category
FROM products
WHERE metadata @> '{"category":"book"}'::jsonb;

SELECT name FROM products WHERE name ILIKE '%pen%';

->> คืนค่าเป็น text ส่วน @> ตรวจการครอบคลุมของ JSONB และ ILIKE ค้นหาโดยไม่สนตัวพิมพ์เล็กใหญ่ตาม locale

CTE แบ่ง query เป็นขั้น

SQL
WITH order_totals AS (
  SELECT product_id, SUM(quantity) AS units
  FROM orders
  GROUP BY product_id
)
SELECT p.name, t.units
FROM order_totals AS t
JOIN products AS p ON p.id = t.product_id
ORDER BY t.units DESC;

บทที่ 7 — Index และการอ่านแผน

วัดผลก่อนและหลังเพิ่ม index

สร้าง index และตรวจ query plan

SQL
CREATE INDEX products_created_at_idx ON products (created_at);
CREATE INDEX products_metadata_gin_idx ON products USING gin (metadata);

EXPLAIN (ANALYZE, BUFFERS)
SELECT id, name FROM products WHERE created_at >= now() - interval '7 days';

B-tree เป็นค่าเริ่มต้น เหมาะกับเงื่อนไขเท่ากับและช่วง; GIN ใช้ได้กับการค้นหา JSONB บางรูปแบบ planner อาจเลือก sequential scan เมื่อข้อมูลน้อยหรือประเมินว่าเร็วกว่า Index มีต้นทุนพื้นที่และการเขียน ข้อควรระวัง: EXPLAIN ANALYZE รันคำสั่งจริง จึงต้องระวังโดยเฉพาะ UPDATE/DELETE

บทที่ 8 — Transaction และ concurrency

ลดสต็อกอย่างปลอดภัยและบันทึก order ร่วมกัน

คำสั่งต้องสำเร็จทั้งชุด

ตัวอย่างนี้หักสต็อกและสร้าง order ใน statement เดียว หากสต็อกไม่พอจะไม่มีแถวถูกเพิ่ม:

SQL
BEGIN;
WITH updated AS (
  UPDATE products
  SET stock = stock - 2
  WHERE sku = 'BK-01' AND stock >= 2
  RETURNING id
)
INSERT INTO orders (product_id, quantity)
SELECT id, 2 FROM updated
RETURNING id, product_id, quantity;
COMMIT;

หาก INSERT ไม่คืนแถว แสดงว่าไม่พบสินค้าหรือสต็อกไม่พอ ในแอปให้ใช้ parameterized query และจัดการข้อผิดพลาดด้วยการ rollback การ UPDATE แบบมีเงื่อนไขช่วยให้การหักสต็อกพร้อมกันตรวจค่าล่าสุดได้

SQL
BEGIN;
UPDATE products SET stock = stock + 1 WHERE sku = 'BK-01';
ROLLBACK; -- ยกเลิกการเปลี่ยนแปลงใน transaction นี้

PostgreSQL ใช้ MVCC; isolation เริ่มต้นคือ Read Committed ซึ่งแต่ละ statement เห็น snapshot ใหม่ สำหรับกฎธุรกิจที่ซับซ้อนอาจต้องใช้ locking หรือ isolation สูงขึ้นและจัดการ retry

บทที่ 9 — สิทธิ์และการเชื่อมต่อ

แยก role สำหรับแอปจากผู้ดูแล

กำหนดสิทธิ์เท่าที่จำเป็น

ตัวอย่างรันด้วยผู้มีสิทธิ์จัดการ role และตารางใน database shop:

SQL
CREATE ROLE shop_app LOGIN;
GRANT CONNECT ON DATABASE shop TO shop_app;
GRANT USAGE ON SCHEMA public TO shop_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON products, orders TO shop_app;
GRANT USAGE ON ALL SEQUENCES IN SCHEMA public TO shop_app;
-- ใน psql ใช้ \password shop_app เพื่อตั้งรหัสผ่าน

สิทธิ์นี้ครอบคลุมตารางและ sequence ที่มีอยู่แล้วเท่านั้น ตารางใหม่ต้องกำหนดสิทธิ์เพิ่มหรือใช้ default privileges ตามเจ้าของ object เก็บ credential ในระบบจัดการความลับ และใช้ TLS ตามสภาพแวดล้อม

บทที่ 10 — Backup และ restore

ทดลองกู้คืนก่อนนำไปใช้จริง

สำรองฐานข้อมูลฝึก

รันบนเครื่องที่มี PostgreSQL client และเชื่อมต่อ database ได้; ระบบจะถามรหัสผ่านตามการตั้งค่า:

Terminal
pg_dump -h localhost -U postgres -d shop -Fc -f shop.dump
createdb -h localhost -U postgres shop_restore
pg_restore -h localhost -U postgres -d shop_restore shop.dump

ไฟล์ custom format ใช้ pg_restore และควรทดสอบกู้คืนบน database แยกเป็นระยะ pg_dump สำรองทีละ database ไม่รวม global roles; กลยุทธ์ production อาจต้องมีวิธีสำรองและกู้คืนอื่นเพิ่มเติม

บทที่ 11 — แบบฝึกหัดและเอกสารอ้างอิง

ตรวจความเข้าใจและต่อยอด

ลองทำด้วยตัวเอง

  1. เพิ่มสินค้า 3 รายการและค้นหารายการที่ stock ต่ำกว่า 5
  2. สร้างรายงานจำนวนที่ขายรวมต่อสินค้า รวมสินค้าที่ยังไม่มี order
  3. เพิ่มคอลัมน์ราคาต่อหน่วยใน orders แล้วอธิบายว่าทำไมควรเก็บราคาขณะซื้อ
  4. เขียน transaction ตัดสต็อกและสร้าง order เมื่อสต็อกเพียงพอ แล้วทดสอบ rollback
  5. สำรองและกู้คืนลง database ใหม่ จากนั้นเปรียบเทียบจำนวนแถว
